import { useRef, useState } from 'react';
import { submitDonation, donationErrorMessage } from './wallet-flow.mjs';
import { parseEthAmount } from './amount.mjs';

/** React UI sample. The caller supplies an injected wallet and contract adapter. */
export function DonationForm({ ethereum, campaignId, sendDonation }) {
  const locked = useRef(false);
  const [amount, setAmount] = useState('');
  const [stage, setStage] = useState('idle');
  const [error, setError] = useState('');
  const [hash, setHash] = useState('');
  const busy = ['connecting', 'switching', 'confirming', 'pending'].includes(stage);

  async function handleSubmit(event) {
    event.preventDefault();
    // A ref guards double submissions before React has committed its next render.
    if (locked.current) return;
    setError('');
    setHash('');
    try { parseEthAmount(amount); }
    catch (validationError) { setError(validationError.message); return; }
    locked.current = true;
    try {
      const receipt = await submitDonation({
        ethereum, campaignId, amount, sendDonation, onStage: setStage,
      });
      setHash(receipt.transactionHash || receipt.hash || '');
      setAmount('');
    } catch (transactionError) {
      setStage('error');
      setError(donationErrorMessage(transactionError));
    } finally {
      locked.current = false;
    }
  }

  return (
    <form onSubmit={handleSubmit} aria-busy={busy}>
      <label htmlFor="donation-amount">Amount in Sepolia ETH</label>
      <input id="donation-amount" inputMode="decimal" value={amount}
        onChange={event => setAmount(event.target.value)} disabled={busy} required
        aria-describedby={error ? 'donation-error' : undefined} />
      <button disabled={busy} type="submit">{busy ? 'Processing…' : 'Contribute'}</button>
      <p role="status" aria-live="polite">{stage === 'idle' ? '' : stage}</p>
      {error && <p role="alert" id="donation-error">{error}</p>}
      {stage === 'success' && <p>Confirmed{hash ? `: ${hash}` : ''}</p>}
    </form>
  );
}

import { parseEthAmount } from './amount.mjs';
export const SEPOLIA_CHAIN_ID = '0xaa36a7';

/**
 * EIP-1193 wallet orchestration. sendDonation is supplied by the application:
 * ({account, campaignId, value}) => contract.donateToCampaign(campaignId, {value})
 * Use an ethers signer bound to the returned account. No private keys are handled.
 */
export async function submitDonation({ ethereum, campaignId, amount, sendDonation, onStage = () => {} }) {
  if (!ethereum?.request) throw new Error('An Ethereum wallet is required.');
  if (typeof campaignId !== 'string' || !/^(0|[1-9]\d{0,19})$/.test(campaignId)) {
    throw new Error('Invalid campaign ID.');
  }
  const value = parseEthAmount(amount).toString();
  onStage('connecting');
  const accounts = await ethereum.request({ method: 'eth_requestAccounts' });
  const account = accounts?.[0];
  if (!/^0x[0-9a-fA-F]{40}$/.test(account || '')) throw new Error('No wallet account selected.');

  const chain = await ethereum.request({ method: 'eth_chainId' });
  if (chain.toLowerCase() !== SEPOLIA_CHAIN_ID) {
    onStage('switching');
    await ethereum.request({ method: 'wallet_switchEthereumChain', params: [{ chainId: SEPOLIA_CHAIN_ID }] });
  }
  // Recheck after switching: the user may reject, switch accounts, or change networks.
  const currentChain = await ethereum.request({ method: 'eth_chainId' });
  const currentAccounts = await ethereum.request({ method: 'eth_accounts' });
  if (currentChain.toLowerCase() !== SEPOLIA_CHAIN_ID) throw new Error('Select Sepolia before continuing.');
  if (currentAccounts?.[0]?.toLowerCase() !== account.toLowerCase()) {
    throw new Error('Wallet account changed. Please submit again.');
  }

  onStage('confirming');
  const transaction = await sendDonation({ account, campaignId, value });
  onStage('pending');
  const receipt = await transaction.wait();
  if (receipt?.status !== 1) throw new Error('The transaction did not succeed.');
  onStage('success');
  return receipt;
}

export function donationErrorMessage(error) {
  const code = error?.code ?? error?.error?.code;
  if (code === 4001 || code === 'ACTION_REJECTED') return 'Request cancelled in your wallet.';
  if (code === 4902) return 'Add the Sepolia network to your wallet, then try again.';
  if (code === 'INSUFFICIENT_FUNDS') return 'Insufficient test ETH for the donation and gas.';
  if (code === 'TRANSACTION_REPLACED') return 'Transaction replaced. Check your wallet before submitting again.';
  return 'Donation could not be completed. Check your wallet and try again.';
}

/** Convert a positive decimal ETH string to wei without floating-point arithmetic. */
export function parseEthAmount(value) {
  if (typeof value !== 'string' || !/^(0|[1-9]\d{0,11})(\.\d{1,18})?$/.test(value)) {
    throw new Error('Enter a decimal ETH amount with up to 18 decimal places.');
  }
  const [whole, fraction = ''] = value.split('.');
  const wei = BigInt(whole) * 10n ** 18n + BigInt(fraction.padEnd(18, '0'));
  if (wei <= 0n) throw new Error('Amount must be greater than zero.');
  return wei;
}

import http from 'node:http';
import { pathToFileURL } from 'node:url';
import { parseEthAmount } from './amount.mjs';

const MAX_BYTES = 8192;
class RequestError extends Error {
  constructor(status, message) { super(message); this.status = status; }
}
async function readJson(request) {
  if (request.headers['content-type']?.split(';')[0].trim() !== 'application/json') {
    throw new RequestError(415, 'Content-Type must be application/json.');
  }
  let bytes = 0;
  const chunks = [];
  for await (const chunk of request) {
    bytes += chunk.length;
    if (bytes > MAX_BYTES) throw new RequestError(413, 'Request body is too large.');
    chunks.push(chunk);
  }
  try { return JSON.parse(Buffer.concat(chunks).toString('utf8')); }
  catch { throw new RequestError(400, 'Malformed JSON.'); }
}
function reply(response, status, body) {
  response.writeHead(status, { 'Content-Type': 'application/json', 'Cache-Control': 'no-store' });
  response.end(JSON.stringify(body));
}

/** A read-only validation API: no payments, accounts, writes, or wallet signatures. */
export function createApiServer() {
  return http.createServer(async (request, response) => {
    try {
      const path = new URL(request.url, 'http://localhost').pathname;
      if (path === '/api/health' && request.method === 'GET') {
        return reply(response, 200, { status: 'ok' });
      }
      if (path !== '/api/contributions/preview') return reply(response, 404, { error: 'Not found.' });
      if (request.method !== 'POST') {
        response.setHeader('Allow', 'POST');
        return reply(response, 405, { error: 'Use POST.' });
      }
      const body = await readJson(request);
      if (!body || typeof body !== 'object' || Array.isArray(body)) {
        throw new RequestError(400, 'Expected a JSON object.');
      }
      if (Object.keys(body).some(key => !['campaignId', 'amount'].includes(key))) {
        throw new RequestError(400, 'Unexpected request fields.');
      }
      if (typeof body.campaignId !== 'string' || !/^(0|[1-9]\d{0,19})$/.test(body.campaignId)) {
        throw new RequestError(400, 'Invalid campaign ID.');
      }
      let wei;
      try { wei = parseEthAmount(body.amount); }
      catch (error) { throw new RequestError(400, error.message); }
      return reply(response, 200, {
        campaignId: body.campaignId, amountWei: wei.toString(), chainId: 11155111,
        previewOnly: true,
      });
    } catch (error) {
      if (!response.destroyed && !response.headersSent) {
        reply(response, error instanceof RequestError ? error.status : 500, {
          error: error instanceof RequestError ? error.message : 'Unable to process request.',
        });
      }
    }
  });
}

if (process.argv[1] && import.meta.url === pathToFileURL(process.argv[1]).href) {
  const port = Number(process.env.PORT || 3001);
  createApiServer().listen(port, '127.0.0.1', () => console.log(`API sample: http://127.0.0.1:${port}`));
}

/** Share concurrent connection attempts; clear failed attempts so a later call retries. */
export function createConnectionManager(connect) {
  let connection;
  let pending;
  return async function getConnection() {
    if (connection) return connection;
    if (!pending) {
      pending = Promise.resolve().then(connect).then(result => {
        connection = result;
        return result;
      }).finally(() => { pending = undefined; });
    }
    return pending;
  };
}
// Adapter example: createConnectionManager(() => mongoose.connect(process.env.MONGODB_URI))
// This handles initial failures. Established-connection recovery belongs to the driver.

# Engineering portfolio samples
Independent portfolio exercises, not previous client engagements. These files are newly written and do not contain the client's private project code.

## Sample 1: React + wallet transaction flow
DonationForm.jsx, wallet-flow.mjs, amount.mjs
- Exact decimal-to-wei conversion, input validation, network/account rechecks.
- Pending states, a synchronous duplicate-submit guard and rejected-transaction handling.
- Injected wallet and contract adapter; no secrets or real transaction UI in the published gallery.
- Integration snippet: sendDonation: async ({account,campaignId,value}) => new ethers.Contract(address,abi,provider.getSigner(account)).donateToCampaign(campaignId,{value})
- Contract rules, gas estimation, wallet disconnect subscriptions and concurrent chain changes during the wallet prompt require application-specific integration. This is not a smart contract audit or a complete payment system.

## Sample 2: Node.js API + retryable connection
api-server.mjs, retryable-connection.mjs
- Run with Node.js 20+: node samples/api-server.mjs
- GET /api/health
- POST /api/contributions/preview with application/json: {"campaignId":"4","amount":"0.1"}
- Validates and converts only. No donations are sent or persisted.
- The connection manager is adapter-based. No MongoDB credentials are included; real database integration is not claimed.
- API runs locally, not on the static portfolio host. Production auth, rate limiting and payment verification are outside this sample.

## Verification
Run: node --test tests/samples.test.mjs
Six automated checks cover precision/validation, network switching, account changes, rejected wallet requests, failed receipts, connection retries and HTTP responses. Wallet tests use mocks. No live blockchain or MongoDB verification has been performed. DonationForm.jsx is provided as reviewable React source; the standalone API and wallet logic run without third-party dependencies.

import test from 'node:test';
import assert from 'node:assert/strict';
import { once } from 'node:events';

import { submitDonation, SEPOLIA_CHAIN_ID } from '../samples/wallet-flow.mjs';
import { createConnectionManager } from '../samples/retryable-connection.mjs';
import { createApiServer } from '../samples/api-server.mjs';
const account = '0x' + '1'.repeat(40);
function wallet({ chain = SEPOLIA_CHAIN_ID, rejectSwitch = false, changedAccount = false } = {}) {
  return { request: async ({method}) => {
    if (method === 'eth_chainId') return chain;
    if (method === 'eth_requestAccounts') return [account];
    if (method === 'eth_accounts') return [changedAccount ? '0x' + '2'.repeat(40) : account];
    if (method === 'wallet_switchEthereumChain') {
      if (rejectSwitch) throw Object.assign(new Error('Rejected'), {code:4001});
      chain = SEPOLIA_CHAIN_ID;
    }
  }};
}
test('amounts preserve wei precision and reject invalid input', () => {
  assert.equal(parseEthAmount('0.000000000000000001'),1n);
  assert.equal(parseEthAmount('1.234567890123456789'),1234567890123456789n);
  for(const invalid of ['0','-1','1e3','NaN','0.0000000000000000001',' 1',1]) {
    assert.throws(() => parseEthAmount(invalid));
  }
});
test('network switching precedes send; success requires receipt', async () => {
  const stages = [];
  const result = await submitDonation({ethereum:wallet({chain:'0x1'}),campaignId:'2',amount:'0.1',onStage:s=>stages.push(s),sendDonation:async(args)=>{
    assert.equal(args.value,'100000000000000000');
    return {wait:async()=>({status:1})};
  }});
  assert.equal(result.status,1);
  assert.deepEqual(stages,['connecting','switching','confirming','pending','success']);
});
